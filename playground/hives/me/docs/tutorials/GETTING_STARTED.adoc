== Getting Started with My Language

=== Installation

==== From Source

[source,bash]
----
git clone https://github.com/Hyperpolymath/me-dialect-playground
cd me-dialect-playground
cargo build --release
cargo install --path .
----

==== Using Cargo

[source,bash]
----
cargo install my-lang
----

=== Your First Program

Create `+hello.solo+`:

[source,solo]
----
fn main() {
    println("Hello, World!");
}
----

Compile and run:

[source,bash]
----
my-lang run hello.solo
----

=== Basic Concepts

==== Variables

[source,solo]
----
let x = 42;           // Immutable
let mut counter = 0;  // Mutable
counter = counter + 1;
----

==== Functions

[source,solo]
----
fn add(a: i32, b: i32) -> i32 {
    a + b  // Last expression is return value
}

fn greet(name: str) {
    println("Hello, " + name + "!");
}
----

==== Control Flow

[source,solo]
----
// If expressions
let result = if x > 0 {
    "positive"
} else {
    "non-positive"
};

// Pattern matching
match value {
    0 => println("zero"),
    1..=9 => println("single digit"),
    _ => println("other")
}

// Loops
for i in 0..10 {
    println(i);
}

while counter < 100 {
    counter = counter + 1;
}
----

==== Data Structures

[source,solo]
----
// Struct
struct Point {
    x: i32,
    y: i32
}

let p = Point { x: 10, y: 20 };

// Enum
enum Status {
    Success,
    Error(str)
}

let s = Status::Success;
----

=== Intermediate Topics

==== Generics

[source,solo]
----
fn first<T>(list: [T]) -> T {
    list[0]
}

struct Pair<T, U> {
    first: T,
    second: U
}
----

==== Traits

[source,solo]
----
trait Drawable {
    fn draw(&self);
}

impl Drawable for Circle {
    fn draw(&self) {
        println("Drawing circle");
    }
}
----

==== Error Handling

[source,solo]
----
enum Result<T, E> {
    Ok(T),
    Err(E)
}

fn divide(a: i32, b: i32) -> Result<i32, str> {
    if b == 0 {
        Result::Err("Division by zero")
    } else {
        Result::Ok(a / b)
    }
}

// Using ? operator
fn parse_and_divide(s: str) -> Result<i32, str> {
    let num = parse_int(s)?;  // Propagates error
    divide(100, num)
}
----

=== Advanced Features

==== Contracts

[source,solo]
----
fn sqrt(x: f64) -> f64
where
    pre x >= 0.0,
    post result * result <= x + 0.001,
    post result * result >= x - 0.001
{
    // Implementation verified against contract
}
----

==== Affine Types

[source,solo]
----
struct File {
    handle: affine FileHandle
}

impl File {
    fn read(self) -> str {
        // Consumes self - can only read once
        self.handle.read_all()
    }
}
----

==== Async/Await

[source,solo]
----
async fn fetch_data(url: str) -> Result<str, Error> {
    let response = http::get(url).await?;
    Ok(response.body())
}

async fn main() {
    let data = fetch_data("https://api.example.com").await;
    println(data);
}
----

==== Comptime

[source,solo]
----
comptime fn factorial(n: u64) -> u64 {
    if n <= 1 { 1 } else { n * factorial(n-1) }
}

const FACT_10: u64 = comptime factorial(10);  // Computed at compile time
----

=== Moving to Duet

Enable AI features:

[source,duet]
----
// Synthesis
@synth(spec: |x| x * 2)
fn double(x: i32) -> i32;

// Verification
@verify(property: |arr| is_sorted(arr))
fn sort(arr: &mut [i32]) {
    arr.sort();
}

// Intent
fn main() {
    let result = intent("find prime numbers under 100", {
        outputs: [i32]
    });
}
----
